//http es un módulo de Node.js que permite crear servidores web. Sirve para recibir solicitudes de los usuarios y enviar respuestas (response) como páginas HTML, archivos o datos en formato JSON import http from 'http'; //fs significa "File System". Es un módulo de Node.js que permite leer, escribir y manejar archivos dentro del servidor. import fs from 'fs'; //Esta función deberá mostrar una página HTML //con la bienvenida a tu proyecto function darBienvenida(req, res) { fs.readFile('bienvenida.html', 'utf8', (error, data) => { if (error) { //El código 500 significa "Internal Server Error" //Indica que ocurrió un error dentro del servidor al intentar procesar la solicitud. res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Oh no!!!!'); return; } //El código 200 significa "OK" //Indica que la solicitud se procesó correctamente y el servidor envía la respuesta. res.writeHead(200, { 'Content-Type': 'text/html' }); res.end(data); }); } //Esta función deberá enviar un json con los datos de las mascotas function getMascotas(req, res) { //Esto representa un objeto JSON con varias mascotas const mascotas = [ { "nombre": "Pikachu", "color": "Amarillo" }, { "nombre": "Bulbasaur", "color": "Verde" } ]; res.writeHead(200, { 'Content-Type': 'application/json' }); //JSON.stringify convierte un objeto de JavaScript en texto JSON. //Es necesario porque las respuestas HTTP solo pueden enviar texto, //no objetos directamente. res.end(JSON.stringify(mascotas)); } function mostrarPerfil(req, res) { fs.readFile('perfil.html', 'utf8', (error, data) => { if (error) { res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Oh no!!!!'); return; } res.writeHead(200, { 'Content-Type': 'text/html' }); res.end(data); }); } function mostrarAdoptantes(req, res) { fs.readFile('adoptantes.html', 'utf8', (error, data) => { if (error) { res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Oh no!!!!'); return; } res.writeHead(200, { 'Content-Type': 'text/html' }); res.end(data); }); } //Esta función deberá enviar un json con los datos de los adoptantes function getAdoptantes(req, res) { const adoptantes = [ { "nombre": "Carlos", "edad": 28 }, { "nombre": "Ana", "edad": 24 } ]; res.writeHead(200, { 'Content-Type': 'application/json' }); res.end(JSON.stringify(adoptantes)); } //Página 404 function manejarRuta404(req, res) { res.writeHead(404, { 'Content-Type': 'text/plain' }); //404 significa que la página o recurso solicitado no existe res.end('Oops 😅 te perdiste... esta página no existe.'); } //Función para mostrar el equipo function mostrarEquipo(req, res) { fs.readFile('equipo.html', 'utf8', (error, data) => { if (error) { res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Error al cargar la página del equipo'); return; } res.writeHead(200, { 'Content-Type': 'text/html' }); res.end(data); }); } //Función para mostrar la opinión function mostrarOpinion(req, res) { fs.readFile('opinion.html', 'utf8', (error, data) => { if (error) { res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Error al cargar la opinión'); return; } res.writeHead(200, { 'Content-Type': 'text/html' }); res.end(data); }); } //createServer crea un servidor web en Node.js //Documentación: https://nodejs.org/api/http.html#httpcreateserveroptions-requestlistener const servidor = http.createServer((req, res) => { const url = req.url; if (url === '/') { darBienvenida(req, res); } else if (url === '/api/mascotas') { getMascotas(req, res); } else if (url === '/api/adoptantes') { getAdoptantes(req, res); } else if (url === '/adoptantes') { mostrarAdoptantes(req, res); } else if (url === '/equipo') { mostrarEquipo(req, res); } else if (url === '/opinion') { mostrarOpinion(req, res); } else { manejarRuta404(req, res); } }); const puerto = 1984; servidor.listen(puerto, () => { console.log(`Servidor escuchando en el puerto ${puerto}`); });
