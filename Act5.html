fs.readFile('bienvenida.html', 'utf8', (error, data) => { if (error) { // El código 500 significa "Internal Server Error". Indica que algo salió mal en el servidor (como que no encontró el archivo). res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Oh no!!!! Error interno en el servidor.'); return; } // El código 200 significa "OK". Indica que la petición fue exitosa y todo está bien. res.writeHead(200, { 'Content-Type': 'text/html' }); res.end(data); }); } // Envía un JSON con los datos de las mascotas function getMascotas(req, res) { // Agregamos otra mascota como pide la actividad const mascotas = [ { "nombre": "Pikachu", "color": "Amarillo" }, { "nombre": "Gary", "color": "Azul/Caracol" } ]; res.writeHead(200, { 'Content-Type': 'application/json' }); // JSON.stringify convierte un objeto de JavaScript en una cadena de texto (string). // Se usa porque el protocolo HTTP solo puede enviar texto o bytes, no objetos vivos de JS. res.end(JSON.stringify(mascotas)); } function mostrarAdoptantes(req, res) { fs.readFile('adoptantes.html', 'utf8', (error, data) => { if (error) { res.writeHead(500, { 'Content-Type': 'text/plain' }); res.end('Oh no!!!!'); return; } res.writeHead(200, { 'Content-Type': 'text/html' }); res.end(data); }); } // Corregimos getAdoptantes para que envíe un JSON real function getAdoptantes(req, res) { const adoptantes = [ { "nombre": "Ash Ketchum", "ciudad": "Pueblo Paleta" }, { "nombre": "Misty", "ciudad": "Ciudad Celeste" } ]; res.writeHead(200, { 'Content-Type': 'application/json' }); res.end(JSON.stringify(adoptantes)); } // Nueva función para mostrar el equipo function mostrarEquipo(req, res) { fs.readFile('equipo.html', 'utf8', (error, data) => { if (error) { res.writeHead(500, { 'Content-Type': 'text/plain' });
